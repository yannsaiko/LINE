if (res) { const cols = res.columns.map(c => c.toUpperCase()); // キー（ID / MID）となり得るカラムを検索 const idIndices = []; cols.forEach((c, i) => { if (/Z_PK|ROWID|^ID$|_ID|ZMID|MID|ADDRESSBOOKID|SERVER_ID|MEMBER_MID/i.test(c)) { idIndices.push(i); } }); // 名前が格納されているカラム（優先度順） const priorityCols = ['ZCUSTOMNAME', 'ZNAME', 'ZADDRESSBOOKNAME', 'ZPHONEBOOKNAME', 'ZFULLNAME', 'CUSTOM_NAME', 'DISPLAY_NAME', 'NAME', 'NICKNAME']; const nameIndices = []; priorityCols.forEach(p => { const idx = cols.indexOf(p); if (idx !== -1) nameIndices.push(idx); }); cols.forEach((c, i) => { if (/NAME|TITLE|NICK/i.test(c) && !nameIndices.includes(i)) nameIndices.push(i); } // 3. トーク一覧取得の改善（1対1トークの相手名前補完を強化） function fetchChatRooms(db, sch, userMap, chatMap, fileId, displayLabel) { if (!sch.msgTable) return []; const query = ` SELECT "${sch.chatCol}", COUNT(*), MAX("${sch.timeCol}") FROM "${sch.msgTable}" WHERE "${sch.chatCol}" IS NOT NULL AND "${sch.chatCol}" != '' GROUP BY "${sch.chatCol}" ORDER BY 3 DESC `; const senderRes = db.exec(senderQuery)[0]; if (senderRes && senderRes.values) { for (const sRow of senderRes.values) { const sId = String(sRow[0]).trim(); if (userMap[sId]) { name = userMap[sId]; break; } } } } catch(e){} } if (!name) name = `トーク部屋 (${chatId})`; const dateObj = parseTimestamp(r[2]); return { fileId, displayLabel, id: chatId, name, count: r[1], lastTime: dateObj ? formatDate(dateObj) : '日時不明' }; }); }
