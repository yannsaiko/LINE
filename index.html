// ファイル名に sqlite や db が含まれていれば強制的にDBとして解析を試みる処理を追加 async function handleFiles(fileList) { if (!SQL) return; const statusEl = document.getElementById('status-msg'); detectedRegistry = { databases: [], walShm: [], media: [], configs: [] }; accounts = []; allDiscoveredChats = []; const totalFiles = fileList.length; for (let i = 0; i < totalFiles; i++) { const file = fileList[i]; const name = file.name.toLowerCase(); const path = file.webkitRelativePath || file.name; const sizeMB = (file.size / 1024 / 1024).toFixed(2); if (i % 20 === 0 || i === totalFiles - 1) { statusEl.innerText = `全ファイルを走査中... (${i + 1} / ${totalFiles} 件完了)`; await new Promise(r => setTimeout(r, 0)); } if (name.endsWith('-wal') || name.endsWith('-shm') || name.endsWith('.wal') || name.endsWith('.shm')) { detectedRegistry.walShm.push({ name: file.name, path, size: `${sizeMB} MB` }); continue; } if (/\.(jpg|jpeg|png|gif|webp|mp4|mov|m4a|aud|aac|mp3|pdf|docx|zip|heic|db-journal)$/i.test(name)) { detectedRegistry.media.push({ name: file.name, path, size: `${sizeMB} MB` }); continue; } // SQLite判定の緩和（拡張子判定も追加） const isSqlite = (await isSqliteHeader(file)) || name.includes('sqlite') || name.endsWith('.db'); if (isSqlite) { try { statusEl.innerText = `データベース構造解析中: ${file.name} (${sizeMB} MB)...`; await new Promise(r => setTimeout(r, 0)); const arrayBuffer = await file.arrayBuffer(); const dbInstance = new SQL.Database(new Uint8Array(arrayBuffer)); const dbInfo = analyzeDatabase(dbInstance); if (dbInfo.msgTable) { buildNameDictionary(dbInstance, dbInfo); const accIndex = accounts.length; const accountObj = { id: accIndex, fileName: file.name, filePath: path, fileSize: `${sizeMB} MB`, db: dbInstance, info: dbInfo }; accounts.push(accountObj); detectedRegistry.databases.push({ name: file.name, path, size: `${sizeMB} MB`, isValidLineDb: true }); extractChatsFromAccount(accountObj); } else { detectedRegistry.databases.push({ name: file.name, path, size: `${sizeMB} MB`, isValidLineDb: false }); } } catch (e) { console.warn(`スキップ (${file.name}):`, e); } } } if (accounts.length === 0) { statusEl.innerHTML = ` 有効な LINE データベースが見つかりませんでした。
※ Line.sqlite または SQLite形式のデータベースが含まれているか確認してください。`; return; } dropZone.style.display = 'none'; statusEl.style.display = 'none'; document.getElementById('app-container').style.display = 'flex'; document.getElementById('btn-print').style.display = 'block'; document.getElementById('btn-reset').style.display = 'block'; renderRegistrySummary(); const selectEl = document.getElementById('account-select'); selectEl.innerHTML = ''; accounts.forEach((acc, index) => { const option = document.createElement('option'); option.value = index; option.textContent = `${acc.fileName} (${acc.fileSize})`; selectEl.appendChild(option); }); currentAccount = accounts[0]; renderChatList(); }
